    var body = ROWS.map(function (r, i) {
      var have = Number(HAVE[r.id] || 0);
      var need = r.need == null ? null : Number(r.need);
      var short = need == null ? null : Math.max(0, need - have);
      var priced = r.cost != null;
      if (need != null && priced) {
        totNeed += need * r.cost; totHave += Math.min(have, need) * r.cost; totBuy += short * r.cost;
      }
      if (!priced) unpriced++;
      var pct = need ? Math.min(100, Math.round(have / need * 100)) : 0;
      var cls = need == null ? "par" : (short > 0 ? "short" : "covered");
      if (FILTER === "short" && cls !== "short") return "";
      if (FILTER === "covered" && cls !== "covered") return "";
      if (FILTER === "prep" && !r.prep) return "";
      if (FILTER === "note" && !r.note) return "";
      var bar = need == null ? "" :
        '<div class="bar ' + (pct >= 100 ? "" : pct > 0 ? "part" : "none") + '"><span style="width:' + pct + '%"></span></div>' +
        '<span class="barlbl">' + tidy(have) + " / " + tidy(need) + " " + esc(r.unit) + "  (" + pct + "%)</span>";
      return '<tr class="' + cls + '" data-id="' + esc(r.id) + '">' +

             if (r1) { r1.prep = !r1.prep; persist(); invRender(); } return; }
    var d = e.target.closest && e.target.closest("[data-del]");
    if (d) { var nm = (ROWS.filter(function (r) { return r.id === d.dataset.del; })[0] || {}).name || "this row";
             if (window.confirm("Delete " + nm + "?")) {
               ROWS = ROWS.filter(function (r) { return r.id !== d.dataset.del; }); persist(); invRender(); }
             return; }

  var LS_SALES = "quatro-sales-v1";
  var SALES = load(LS_SALES, {});
  var WEEK0 = new Date(2026, 7, 27); /* Thursday 27.08.2026, the first counted week */
  function weekKey(d) {
    var ms = d - WEEK0, w = Math.floor(ms / (7 * 864e5));
    return w < 0 ? 0 : w;
  }
  function weekLabel(w) {
    var a = new Date(WEEK0.getTime() + w * 7 * 864e5), b = new Date(a.getTime() + 6 * 864e5);
    function f(x) { return ("0" + x.getDate()).slice(-2) + "." + ("0" + (x.getMonth() + 1)).slice(-2) + "." + String(x.getFullYear()).slice(-2); }
    return f(a) + " to " + f(b);
  }
  var CURW = weekKey(new Date());
  function salesHTML() {
    var wk = SALES["w" + CURW] || {};

    var rows = BK.map(function (d) {

      return '<tr><td><span class="slot"><span class="slot-dot" style="background:' + d.sw +

        '<td><button class="sl-b" type="button" data-sales="-|' + esc(d.id) + '">&minus;</button>' +

        '<button class="sl-b" type="button" data-sales="+|' + esc(d.id) + '">+</button></td>' +
        '<td class="num">' + (total ? Math.round(n / total * 100) : 0) + "%</td></tr>";
    }).join("");
    var best = BK.slice().sort(function (a, b) { return (wk[b.id] || 0) - (wk[a.id] || 0); })[0];
