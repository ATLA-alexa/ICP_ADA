  function dayDot(v) {
    var d = String(v || "").replace(/\D/g, "");
    if (!d) return null;
    if (d.length !== 6 && d.length !== 8) return false;
    var day = +d.slice(0, 2), mon = +d.slice(2, 4);
    var yr = d.length === 6 ? 2000 + +d.slice(4, 6) : +d.slice(4, 8);
    if (mon < 1 || mon > 12 || day < 1 || day > 31) return false;
    var x = new Date(yr, mon - 1, day);
    if (x.getDate() !== day || x.getMonth() !== mon - 1) return false;   /* 31.02 and friends */
    var wd = (x.getDay() + 6) % 7;                                       /* JS counts from Sunday */
    var p2 = function (n) { return ("0" + n).slice(-2); };
    return (isNl() ? DD_NL : DD_EN)[wd] + " " + p2(day) + "." + p2(mon) + "." + String(yr).slice(-2);
  }

  function dec(s) { return (typeof LANG !== "undefined" && LANG === "nl") ? String(s).replace(".", ",") : String(s); }

    return dec(v.toFixed(4).replace(/0+$/, "").replace(/\.$/, ""));

  /* Accept both decimal separators. The field is type=text on purpose: a
     type=number input silently DROPS a comma, so "1,4" arrives as 14 and the
     count is ten times out with nothing on screen to show it went wrong. */
  function parseQty(v) {
    v = String(v == null ? "" : v).trim().replace(",", ".");
    if (v === "") return undefined;
    var n = parseFloat(v);
    return isNaN(n) ? undefined : Math.max(0, n);
  }

  /* One source of truth for a row's derived numbers. invHTML draws from it and
     the in-place updater recomputes from it, so they can never drift. */
  function calcRow(r) {
    var have = Number(HAVE[r.id] || 0);
    var need = r.need == null ? null : Number(r.need);
    var short = need == null ? null : Math.max(0, need - have);
    return { have: have, need: need, short: short, priced: r.cost != null,
             pct: need ? Math.min(100, Math.round(have / need * 100)) : 0,
             cls: need == null ? "par" : (short > 0 ? "short" : "covered") };
  }
  function invTotals() {
    var need = 0, have = 0, buy = 0;
    ROWS.forEach(function (r) {
      var c = calcRow(r);
      if (c.need != null && c.priced) {
        need += c.need * r.cost;
        have += Math.min(c.have, c.need) * r.cost;
        buy  += c.short * r.cost;
      }
    });
    return { need: need, have: have, buy: buy };
  }

  /* A full re-render is fine for structural changes, but it must not throw the
     reader back to the top of a long table. */
  function invRender() {
    var y = window.scrollY || document.documentElement.scrollTop || 0;
    if (document.getElementById("root")) render();
    window.scrollTo(0, y);

  /* Typing a stock count repaints only the cells that actually changed. No
     re-render, so the page does not jump and the caret stays where it is. */
  function refreshRow(id) {
    var tr = document.querySelector('tr[data-id="' + id + '"]');
    var r = ROWS.filter(function (x) { return x.id === id; })[0];
    if (!tr || !r) return;
    var c = calcRow(r);
    tr.className = c.cls;

    var bar = tr.querySelector(".bar"), lbl = tr.querySelector(".barlbl");
    if (bar) {
      bar.className = "bar " + (c.pct >= 100 ? "" : c.pct > 0 ? "part" : "none");
      if (bar.firstChild) bar.firstChild.style.width = c.pct + "%";
    }
    if (lbl) lbl.textContent = tidy(c.have) + " / " + tidy(c.need) + " " +
      unitLabel(r.unit) + "  (" + c.pct + "%)";

    var buy = tr.querySelector('[data-c="buy"]');
    if (buy) {
      buy.className = "buy" + (c.short === 0 ? " ok" : "");
      buy.textContent = c.need == null ? tt(SETPAR)
        : c.short > 0 ? tidy(c.short) + " " + unitLabel(r.unit)
        : tt({ en: "covered", nl: "gedekt" });
    }
    var cost = tr.querySelector('[data-c="cost"]');
    if (cost) {
      cost.innerHTML = !c.priced
        ? '<span class="iv-tbd">' + et({ en: "price TBD", nl: "prijs volgt" }) + "</span>"
        : c.short > 0 ? n2(c.short * r.cost) : "-";
    }
    refreshTotals();
  }

  document.addEventListener("blur", function (e) {
    var el = e.target;
    if (!el.dataset || !el.dataset.f) return;
    var tr = el.closest("tr"); if (!tr) return;
    var row = ROWS.filter(function (r) { return r.id === tr.dataset.id; })[0]; if (!row) return;
    var val = el.textContent.trim();
    /* Re-render only on a real change. Blurring a field you merely looked at
       should not rebuild the table under the reader. */
    if (el.dataset.f === "need") {
      var nv = isSetPar(val) ? null : parseQty(val);
      if (nv === undefined) nv = row.need;
      if (nv === row.need) return;
      row.need = nv;
    } else if (el.dataset.f === "name") {
      if (!val) return;
      if (isNl()) { if (val === row.nameNl) return; row.nameNl = val; }
      else { if (val === row.name) return; row.name = val; }
    } else if (el.dataset.f === "note") {
      if (isNl()) { if (val === (row.noteNl || "")) return; row.noteNl = val; }
      else { if (val === (row.note || "")) return; row.note = val; }
    } else if (el.dataset.f === "supplier") {
      var sup = val || "TBD";
      if (sup === row.supplier) return;
      row.supplier = sup;
      /* The shop decides the price, so re-read it from the book. */
      var e = rowEntry(row);
      if (e) row.cost = costIn(e, row.unit);
    }
    persist(); invRender();
  }, true);

      t(d.tbl).map(function (s) { return "<li>" + esc(s) + "</li>"; }).join("") + "</ol></div>";

  /* The entry a row actually buys: the one at the row's own supplier if that
     shop stocks it, otherwise whoever does. */
  function rowEntry(r) {
    var key = ROW_PRICE[r.key];
    if (!key) return null;
    var alts = alternatives(key, r.unit);
    if (!alts.length) return null;
    var mine = alts.filter(function (a) { return a.e.sup === r.supplier; })[0];
    return (mine || alts[0]).e;
  }
  /* The price book is the source of truth wherever it has a line, so every
     number on the Inventory tab can be traced back to a pack and a shop.
     On the seed it also sets the supplier. On saved rows it does not: someone
     may have changed where a line is bought, and that choice decides the
     price, not the other way round. */
  function applyPrices() {
    SEED.forEach(function (row) {
      var e = rowEntry(row);
      if (e) { row.supplier = e.sup; row.cost = costIn(e, row.unit); }
    });
    ROWS.forEach(function (row) {
      var e = rowEntry(row);
      if (e) row.cost = costIn(e, row.unit);
    });
  }

  function rowCheaper(r) {
    var key = ROW_PRICE[r.key];
    if (!key) return null;
    var alts = alternatives(key, r.unit);
    if (alts.length < 2) return null;
    var used = rowEntry(r), best = alts[0].e;
    if (!used || best.id === used.id) return null;
    var uc = costIn(used, r.unit), bc = alts[0].c;
    if (bc >= uc) return null;
    /* Judge the swap on what it saves in a week, not on the per-unit gap. A
       tenth of a cent a gram is noise on 60 g of thyme and real money on 450 g
       of mint. */
    var week = (uc - bc) * (r.need == null ? 0 : Number(r.need));
    return week >= 0.25 ? { e: best, c: bc, week: week } : null;
  }

  /* Convert a price book entry into the unit the inventory row counts in.
     Base units are ml, g and pcs; a row counting in L or kg scales by 1000.
     Anything that cannot be converted, litres against grams say, returns null
     rather than a number that looks right and is not. */
  function unitBase(u) {
    if (u === "L") return { b: "ml", f: 1000 };
    if (u === "kg") return { b: "g", f: 1000 };
    if (u === "g") return { b: "g", f: 1 };
    if (u === "ml") return { b: "ml", f: 1 };
    return { b: "pcs", f: 1 };          /* pcs, cans, bottles, jars, packs, sets */
